# M1.T1.3 — 도메인 CRUD API (5 endpoints)

## Goal

`/api/domains` 5개 엔드포인트(`GET 목록`, `POST 등록`, `GET 단건`, `PATCH 수정`, `DELETE 삭제`)와 Pydantic v2 스키마, 단위 테스트 구현. **인증은 T1.4에서 결선 — 본 Task의 placeholder는 fail-safe 503 반환**으로 dev/staging 무인증 노출 위험 제거. 테스트는 `app.dependency_overrides`로 우회.

> **배포 주의**: T1.4 머지 전엔 `/api/domains/*` 모든 호출이 503("auth not configured")을 반환합니다. 정상이며, T1.4 머지 후 401(미인증) 또는 200(세션 통과)으로 바뀝니다. T1.3만 단독으로 dev/staging에 배포해도 무인증 노출은 발생하지 않습니다.

## Spec

### 엔드포인트 (`docs/design/04_incremental_admin.md` §4.5)

| Method | Path | 본문/응답 |
|---|---|---|
| GET | `/api/domains` | `[DomainOut]` |
| POST | `/api/domains` | `DomainCreate` → `DomainOut` (201) |
| GET | `/api/domains/{id}` | `DomainOut` (없으면 404) |
| PATCH | `/api/domains/{id}` | `DomainUpdate` → `DomainOut` |
| DELETE | `/api/domains/{id}` | 204 (연관 chunks·KB 삭제는 M3.T3.3에서 결선; 본 Task는 PG 삭제만) |

> `DELETE`는 PG cascade로 `pages`/`attachments`/`chunks` 삭제됨(`ondelete="CASCADE"`). Qdrant 삭제는 M3에서 hook 추가.

### `api/schemas/domain.py`

```python
from __future__ import annotations
from datetime import datetime
from decimal import Decimal

from pydantic import BaseModel, ConfigDict, Field, field_validator


class DomainBase(BaseModel):
    host: str = Field(min_length=3, max_length=253)
    scheme: str = Field(default="https", pattern=r"^https?$")
    display_name: str | None = None
    max_depth: int = Field(default=10, ge=0, le=50)
    max_pages: int = Field(default=20_000, ge=1, le=1_000_000)
    rate_limit_rps: Decimal = Field(default=Decimal("1.5"), ge=Decimal("0.1"), le=Decimal("20"))
    concurrency: int = Field(default=4, ge=1, le=32)
    use_playwright: bool = False
    respect_robots: bool = True
    user_agent: str | None = None
    allow_patterns: list[str] | None = None
    deny_patterns: list[str] | None = None
    selector_blacklist: list[str] | None = None
    incremental_cron: str | None = "0 3 * * 0"
    incremental_enabled: bool | None = True
    is_active: bool = True

    @field_validator("host")
    @classmethod
    def host_lowercase(cls, v: str) -> str:
        v = v.strip().lower()
        if v.startswith("http://") or v.startswith("https://"):
            raise ValueError("host should not include scheme")
        if "/" in v:
            raise ValueError("host should not include path")
        return v


class DomainCreate(DomainBase):
    pass


class DomainUpdate(BaseModel):
    # PATCH — 모든 필드 optional
    model_config = ConfigDict(extra="forbid")

    display_name: str | None = None
    max_depth: int | None = Field(default=None, ge=0, le=50)
    max_pages: int | None = Field(default=None, ge=1, le=1_000_000)
    rate_limit_rps: Decimal | None = Field(default=None, ge=Decimal("0.1"), le=Decimal("20"))
    concurrency: int | None = Field(default=None, ge=1, le=32)
    use_playwright: bool | None = None
    respect_robots: bool | None = None
    user_agent: str | None = None
    allow_patterns: list[str] | None = None
    deny_patterns: list[str] | None = None
    selector_blacklist: list[str] | None = None
    incremental_cron: str | None = None
    incremental_enabled: bool | None = None
    is_active: bool | None = None


class DomainOut(DomainBase):
    model_config = ConfigDict(from_attributes=True)

    id: int
    created_at: datetime
    updated_at: datetime
```

### `api/deps.py`

```python
from __future__ import annotations
from collections.abc import AsyncIterator

from fastapi import Depends, HTTPException
from sqlalchemy.ext.asyncio import AsyncSession

from kspo_ingest.db.session import async_session


async def db_session() -> AsyncIterator[AsyncSession]:
    async with async_session() as s:
        yield s


# T1.4에서 실제 세션 인증으로 교체. 본 Task는 fail-safe placeholder —
# T1.4 머지 전 dev/staging에 배포돼도 무인증 노출이 발생하지 않도록 503 반환.
async def require_admin() -> str:
    raise HTTPException(
        status_code=503,
        detail="auth_not_configured",  # T1.4 머지 후 401/세션 통과로 바뀜
    )


SessionDep = Depends(db_session)
AdminDep = Depends(require_admin)
```

### `api/routers/domains.py`

```python
from __future__ import annotations
from datetime import datetime, timezone

from fastapi import APIRouter, HTTPException, Response, status
from sqlalchemy import select
from sqlalchemy.exc import IntegrityError
from sqlalchemy.ext.asyncio import AsyncSession

from kspo_ingest.api.deps import AdminDep, SessionDep
from kspo_ingest.api.schemas.domain import DomainCreate, DomainOut, DomainUpdate
from kspo_ingest.db.models import Domain

router = APIRouter(prefix="/api/domains", tags=["domains"])


@router.get("", response_model=list[DomainOut])
async def list_domains(s: AsyncSession = SessionDep, _admin=AdminDep):
    rows = (await s.execute(select(Domain).order_by(Domain.id))).scalars().all()
    return rows


@router.post("", response_model=DomainOut, status_code=status.HTTP_201_CREATED)
async def create_domain(body: DomainCreate, s: AsyncSession = SessionDep, _admin=AdminDep):
    d = Domain(**body.model_dump())
    s.add(d)
    try:
        await s.commit()
    except IntegrityError:
        await s.rollback()
        raise HTTPException(409, "host already exists") from None
    await s.refresh(d)
    return d


@router.get("/{id}", response_model=DomainOut)
async def get_domain(id: int, s: AsyncSession = SessionDep, _admin=AdminDep):
    d = await s.get(Domain, id)
    if d is None:
        raise HTTPException(404, "domain not found")
    return d


@router.patch("/{id}", response_model=DomainOut)
async def update_domain(id: int, body: DomainUpdate, s: AsyncSession = SessionDep, _admin=AdminDep):
    d = await s.get(Domain, id)
    if d is None:
        raise HTTPException(404, "domain not found")
    patch = body.model_dump(exclude_unset=True)
    for k, v in patch.items():
        setattr(d, k, v)
    d.updated_at = datetime.now(timezone.utc)
    await s.commit()
    await s.refresh(d)
    return d


@router.delete("/{id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_domain(id: int, s: AsyncSession = SessionDep, _admin=AdminDep):
    d = await s.get(Domain, id)
    if d is None:
        raise HTTPException(404, "domain not found")
    await s.delete(d)
    await s.commit()
    # TODO(M3.T3.3): Qdrant payload `domain_id={id}` 삭제 hook
    return Response(status_code=status.HTTP_204_NO_CONTENT)
```

### 테스트 (`test_domains_api.py`)

`httpx.AsyncClient(transport=ASGITransport(app=app))` + pytest-postgresql. **인증 placeholder(503)는 `app.dependency_overrides`로 우회** — T1.3 단위 테스트는 도메인 CRUD 로직만 검증, 인증은 T1.4 단위 테스트에서.

```python
# conftest.py
@pytest.fixture
async def client(app, pg):
    from kspo_ingest.api.deps import require_admin
    app.dependency_overrides[require_admin] = lambda: "test-admin"
    async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as c:
        yield c
    app.dependency_overrides.clear()


# test_domains_api.py
@pytest.mark.asyncio
async def test_placeholder_blocks_unauthed_calls(app):
    """T1.4 머지 전 fail-safe 검증: dependency_overrides 없이 호출하면 503."""
    async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as c:
        r = await c.get("/api/domains")
    assert r.status_code == 503
    assert r.json()["detail"] == "auth_not_configured"


@pytest.mark.asyncio
async def test_list_empty(client):
    r = await client.get("/api/domains")
    assert r.status_code == 200 and r.json() == []


@pytest.mark.asyncio
async def test_create_get_delete(client):
    r = await client.post("/api/domains", json={"host": "www.kspo.or.kr", "display_name": "KSPO"})
    assert r.status_code == 201
    did = r.json()["id"]

    r = await client.get(f"/api/domains/{did}")
    assert r.status_code == 200 and r.json()["host"] == "www.kspo.or.kr"

    r = await client.delete(f"/api/domains/{did}")
    assert r.status_code == 204

    r = await client.get(f"/api/domains/{did}")
    assert r.status_code == 404


@pytest.mark.asyncio
async def test_create_duplicate_host_409(client):
    payload = {"host": "dup.example.com"}
    await client.post("/api/domains", json=payload)
    r = await client.post("/api/domains", json=payload)
    assert r.status_code == 409


@pytest.mark.asyncio
async def test_patch_partial_update(client):
    r = await client.post("/api/domains", json={"host": "patch.example.com"})
    did = r.json()["id"]
    r = await client.patch(f"/api/domains/{did}", json={"max_depth": 3})
    assert r.status_code == 200 and r.json()["max_depth"] == 3
    # 다른 필드는 변경되지 않음
    assert r.json()["max_pages"] == 20_000


@pytest.mark.asyncio
async def test_invalid_host_rejects_scheme(client):
    r = await client.post("/api/domains", json={"host": "https://example.com"})
    assert r.status_code == 422


@pytest.mark.asyncio
async def test_get_404(client):
    r = await client.get("/api/domains/99999")
    assert r.status_code == 404
```
