# M4.T4.3 — 주간 cron (`weekly_incremental_all`) + 통계 뷰 (`refresh_stats_views`)

## Goal

ARQ cron 2종 등록:
1. **`weekly_incremental_all`** — 매주 일요일 03:00 KST에 활성 도메인(`is_active=true`, `incremental_enabled=true`) 모두를 incremental Job으로 enqueue.
2. **`refresh_stats_views`** — 5분 주기로 PG 머티리얼라이즈드 뷰(`v_stats_overview`) 갱신.

PG 통계 뷰 2종(`v_stats_overview` 머티리얼라이즈드, `v_cost_daily` 일반 view)도 본 Task에서 마이그레이션으로 추가. **M4 종료 Task**.

### Migration `0003_stats_views`

```python
revision = "0003_stats_views"
down_revision = "0002_core_tables"


def upgrade() -> None:
    # 1) v_stats_overview — 도메인별 INDEXED/DROPPED/ERROR/총 토큰
    op.execute("""
        CREATE MATERIALIZED VIEW v_stats_overview AS
        SELECT d.id AS domain_id, d.host,
               COUNT(p.*) FILTER (WHERE p.status='INDEXED')        AS indexed,
               COUNT(p.*) FILTER (WHERE p.status='DROPPED')        AS dropped,
               COUNT(p.*) FILTER (WHERE p.status='ERROR')          AS errors,
               COUNT(p.*) FILTER (WHERE p.status='REMOVED')        AS removed,
               COALESCE(SUM(c.token_len), 0)                       AS total_tokens
          FROM domains d
          LEFT JOIN pages p  ON p.domain_id=d.id
          LEFT JOIN chunks c ON c.page_id=p.id
         GROUP BY d.id, d.host;
    """)
    op.execute("CREATE UNIQUE INDEX idx_v_stats_overview_domain ON v_stats_overview(domain_id);")
    # UNIQUE INDEX 필요 — REFRESH MATERIALIZED VIEW CONCURRENTLY를 위해

    # 2) v_cost_daily — 모델별·purpose별 일별 비용 합계 (일반 view)
    op.execute("""
        CREATE VIEW v_cost_daily AS
        SELECT date_trunc('day', created_at) AS day,
               model, purpose, SUM(cost_usd) AS cost_usd
          FROM llm_usage
         GROUP BY 1, 2, 3
         ORDER BY 1 DESC;
    """)


def downgrade() -> None:
    op.execute("DROP VIEW IF EXISTS v_cost_daily;")
    op.execute("DROP INDEX IF EXISTS idx_v_stats_overview_domain;")
    op.execute("DROP MATERIALIZED VIEW IF EXISTS v_stats_overview;")
```

> `v_cost_daily`는 일반 view (실시간) — `llm_usage` 행 수가 적어 매번 집계해도 비용 낮음. `v_stats_overview`는 머티리얼라이즈드 — 페이지·청크 수가 많아 매번 집계는 비용 발생.
> KRW 환산은 view에서 안 함 — 환율 변동 시 환산 시점이 모호해짐. UI/리포트 측에서 `cost_usd × USD_TO_KRW_RATE` 적용.

### ARQ Cron (`tasks/cron.py`)

```python
from arq.cron import cron
from datetime import time as time_t


async def weekly_incremental_all(ctx) -> None:
    """매주 일요일 03:00 KST. 활성·incremental_enabled 도메인 모두 enqueue."""
    s = ctx["session_factory"]()
    domains = (await s.execute(
        select(Domain).where(Domain.is_active == True, Domain.incremental_enabled == True)
    )).scalars().all()

    pool = ctx["arq_pool"]
    enqueued = 0
    for d in domains:
        # 동일 도메인의 진행 중 Job 검사 — RUNNING이면 skip (중복 방지)
        active = await s.scalar(
            select(func.count()).select_from(CrawlJob).where(
                CrawlJob.domain_id == d.id,
                CrawlJob.status.in_(["PENDING", "RUNNING"]),
            )
        )
        if active > 0:
            logger.info("weekly_skip_active", domain_id=d.id, host=d.host)
            continue

        job = CrawlJob(domain_id=d.id, job_type="incremental", status="PENDING",
                       triggered_by="cron", stats={"discovered": 0, ...})
        s.add(job); await s.commit(); await s.refresh(job)
        await pool.enqueue_job("run_incremental", job_id=job.id, domain_id=d.id)
        enqueued += 1

    logger.info("weekly_incremental_all", domains=len(domains), enqueued=enqueued)


async def refresh_stats_views(ctx) -> None:
    """5분 주기 — v_stats_overview 머티리얼라이즈드 뷰 REFRESH CONCURRENTLY."""
    s = ctx["session_factory"]()
    try:
        await s.execute(text("REFRESH MATERIALIZED VIEW CONCURRENTLY v_stats_overview;"))
        await s.commit()
    except Exception as e:
        # CONCURRENTLY는 첫 호출 전 일반 REFRESH 1회 필요 — fallback
        logger.warning("stats_refresh_fallback", error=str(e))
        await s.rollback()
        await s.execute(text("REFRESH MATERIALIZED VIEW v_stats_overview;"))
        await s.commit()
```

`WorkerSettings.cron_jobs`:

```python
cron_jobs = [
    cron(weekly_incremental_all,
         hour={3}, minute={0}, weekday="sun",  # 03:00 KST 일요일
         timezone="Asia/Seoul"),
    cron(refresh_stats_views,
         minute={0, 5, 10, 15, 20, 25, 30, 35, 40, 45, 50, 55}),  # 5분 주기
]
```

> ARQ cron의 timezone 인자는 라이브러리 버전에 따라 다름. `arq>=0.26`은 `pytz` 또는 `zoneinfo` 객체 받음 — `from zoneinfo import ZoneInfo; tz=ZoneInfo("Asia/Seoul")` 사용.
