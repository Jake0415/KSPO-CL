### KPI 측정 스크립트 (`scripts/kpi_measure.py`)

```python
#!/usr/bin/env python
"""
사용:
    python scripts/kpi_measure.py [--domain-id N] [--days 7] [--json]

산출:
    1) 적재 커버리지 — INDEXED/총 페이지(REMOVED 제외)
    2) 적재 신선도 — last_indexed_at - last_fetched_at 평균
    3) Job 성공률 — 최근 N일 SUCCESS/총 Job
    4) 월 LLM 비용 — 이번 달 누계 KRW
"""
async def measure(domain_id: int | None, days: int) -> KpiResult: ...


@dataclass
class KpiResult:
    coverage_pct: float        # INDEXED / (INDEXED + DROPPED + ERROR)
    avg_freshness_days: float  # 페이지의 last_indexed_at - posted_at 평균 (posted_at 없으면 last_fetched_at - discovered_at)
    job_success_rate_pct: float
    monthly_cost_krw: Decimal
    measured_at: datetime
    domain_filter: int | None  # None이면 전체
```

```python
def main():
    import argparse
    p = argparse.ArgumentParser()
    p.add_argument("--domain-id", type=int)
    p.add_argument("--days", type=int, default=7)
    p.add_argument("--json", action="store_true")
    args = p.parse_args()

    r = asyncio.run(measure(args.domain_id, args.days))
    if args.json:
        print(json.dumps(asdict(r), indent=2, default=str))
    else:
        print(f"적재 커버리지: {r.coverage_pct:.1f}%")
        print(f"평균 신선도: {r.avg_freshness_days:.1f}일")
        print(f"Job 성공률(최근 {args.days}일): {r.job_success_rate_pct:.1f}%")
        print(f"이번 달 LLM 비용: {r.monthly_cost_krw}원")
```
