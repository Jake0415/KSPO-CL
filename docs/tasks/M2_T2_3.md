# M2.T2.3 — Refiner (사전 룰 + OpenAI structured output + llm_usage)

## Goal

T2.2의 cleaned 본문을 받아 (1) 사전 룰 3종 — `too_short` / `error_page` / `non_korean` 검사 → DROPPED 결정. 통과 페이지만 (2) OpenAI structured output(4 필드: refined_text/is_meaningful/summary/keywords) 호출 → (3) `llm_usage` 행 기록. **HTTPS_PROXY** 환경변수가 자동 적용되어 사내 프록시 경유 호출.

## Spec

### 사전 룰 (`pre_rules.py`) — `02_pipeline.md` §2.12

```python
HANGUL_RE = re.compile(r"[가-힣]")  # 한글 음절
LETTER_RE = re.compile(r"[^\W\d_]", re.UNICODE)


@dataclass
class PreCheckResult:
    drop: bool
    reason: Literal["too_short", "error_page", "non_korean"] | None


def check(body_text: str, title: str | None, *, ko_ratio_threshold: float) -> PreCheckResult:
    # 1) too_short: len < 100자
    if len(body_text) < 100:
        return PreCheckResult(drop=True, reason="too_short")

    # 2) error_page: title 정확 일치 (대소문자 무관, 공백 trim)
    ERROR_TITLES = {"로그인", "오류", "404", "page not found", "에러", "오류 페이지"}
    if title and title.strip().lower() in {t.lower() for t in ERROR_TITLES}:
        return PreCheckResult(drop=True, reason="error_page")

    # 3) non_korean: 한글 음절 비율 < threshold
    letters = LETTER_RE.findall(body_text)
    if letters:
        hangul = HANGUL_RE.findall(body_text)
        ratio = len(hangul) / len(letters)
        if ratio < ko_ratio_threshold:
            return PreCheckResult(drop=True, reason="non_korean")

    return PreCheckResult(drop=False, reason=None)
```

SYSTEM_PROMPT = """한국어 콘텐츠 정제 전문가.

(1) 정제: 머리글/꼬리글/광고/내비 잔여 노이즈 제거. 표·리스트는 자연어로.
    원문에 없는 정보 추가 금지(환각 방지).

(2) 의미 판별: 검색·답변에 인용될 정보가 충분한지 판단.
    의미 없음: 빈/로그인/에러/메뉴/사이트맵, 본문 < 100자.

(3) 요약: 2~4문장.

(4) 키워드: 5~15개. 고유명사·기관명·법령명·날짜 우선.

응답은 지정된 JSON Schema로만."""


def build_user_prompt(title: str | None, body_text: str) -> str:
    return f"제목: {title or '(없음)'}\n\n본문:\n{body_text}"

### OpenAI 클라이언트 (`openai_client.py`)

```python
from openai import AsyncOpenAI
from tenacity import AsyncRetrying, stop_after_attempt, wait_exponential, retry_if_exception_type
from openai import APIConnectionError, APITimeoutError, RateLimitError, APIStatusError


# 모듈 레벨 singleton — settings에서 키 읽음. proxy는 httpx가 HTTPS_PROXY 환경변수 자동 인식.
_client: AsyncOpenAI | None = None


def get_client() -> AsyncOpenAI:
    global _client
    if _client is None:
        _client = AsyncOpenAI(api_key=settings.openai_api_key, timeout=60.0)
    return _client


# 5xx/429/네트워크 오류만 재시도. 401/400은 즉시 raise (런북 B.2.A/B.2.B 참고).
RETRYABLE = (APIConnectionError, APITimeoutError, RateLimitError)


async def call_with_retry(coro_factory):
    async for attempt in AsyncRetrying(
        stop=stop_after_attempt(5),
        wait=wait_exponential(multiplier=1, min=1, max=300),
        retry=retry_if_exception_type(RETRYABLE),
        reraise=True,
    ):
        with attempt:
            return await coro_factory()
```

### Refiner 메인 (`refiner.py`)

```python
@dataclass
class RefineResult:
    drop: bool
    drop_reason: str | None
    refined: RefinerOutput | None
    usage: TokenUsage | None     # prompt/completion/total
    model: str


async def refine_page(*, title: str | None, body_text: str) -> RefineResult:
    pre = pre_rules.check(body_text, title, ko_ratio_threshold=settings.ko_ratio_threshold)
    if pre.drop:
        return RefineResult(drop=True, drop_reason=pre.reason, refined=None, usage=None,
                            model=settings.openai_refine_model)

    client = get_client()
    model = settings.openai_refine_model
    response = await call_with_retry(lambda: client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user",   "content": build_user_prompt(title, body_text)},
        ],
        response_format={"type": "json_schema", "json_schema": REFINER_JSON_SCHEMA},
        temperature=0.0,
    ))

    raw = response.choices[0].message.content
    refined = RefinerOutput.model_validate_json(raw)
    usage = TokenUsage(
        prompt_tokens=response.usage.prompt_tokens,
        completion_tokens=response.usage.completion_tokens,
        total_tokens=response.usage.total_tokens,
    )

    # is_meaningful=False면 사후 DROPPED 결정 (호출자가 처리)
    return RefineResult(drop=False, drop_reason=None, refined=refined, usage=usage, model=model)
```

> **로그**: `event=refine_ok page_id=... model=... prompt=NNN completion=NNN cost_krw=NN.NN`. 본문 텍스트 출력 금지. `event=refine_drop reason=too_short|error_page|non_korean page_id=...`도 별도 1줄.

### 비용 추적 (`cost.py`)

```python
# OpenAI 가격 (USD per 1M tokens). 모델 추가 시 PR로 갱신 (Tier 2: 자동 회수).
PRICING_USD_PER_1M = {
    # gpt-5.4-mini (예시 가격 — 실제는 OpenAI 페이지 확인 후 갱신)
    "gpt-5.4-mini":          {"prompt": 0.150, "completion": 0.600},
    # 임베딩 모델은 cost.py에서 함께 관리하나 본 Task는 refine만
}


def calc_cost_usd(model: str, prompt_tokens: int, completion_tokens: int) -> Decimal:
    rate = PRICING_USD_PER_1M.get(model)
    if rate is None:
        return Decimal("0")  # 미등록 모델은 0으로 — WARN 로그
    cost = (Decimal(prompt_tokens) * Decimal(str(rate["prompt"])) +
            Decimal(completion_tokens) * Decimal(str(rate["completion"]))) / Decimal("1000000")
    return cost.quantize(Decimal("0.000001"))


async def record_llm_usage(
    session: AsyncSession, *,
    page_id: int | None, attachment_id: int | None, job_id: int | None,
    purpose: Literal["refine", "embedding"], model: str, usage: TokenUsage,
) -> None:
    cost_usd = calc_cost_usd(model, usage.prompt_tokens, usage.completion_tokens)
    row = LlmUsage(
        page_id=page_id, attachment_id=attachment_id, job_id=job_id,
        purpose=purpose, model=model,
        prompt_tokens=usage.prompt_tokens,
        completion_tokens=usage.completion_tokens,
        total_tokens=usage.total_tokens,
        cost_usd=cost_usd,
    )
    session.add(row)
    await session.commit()
```
