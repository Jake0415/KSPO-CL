### KPI 측정 코어 (`observability/kpi.py`)

T5.2의 스크립트 로직을 모듈화 — cron + API + script 모두 동일 함수 사용.

```python
@dataclass
class KpiMeasurement:
    domain_id: int | None              # None = 전체
    measured_at: datetime
    coverage_pct: Decimal
    avg_freshness_days: Decimal | None
    job_success_rate_pct: Decimal
    monthly_cost_krw: Decimal
    indexed_count: int
    dropped_count: int
    error_count: int


async def measure_kpi(s: AsyncSession, *, domain_id: int | None, days: int = 7) -> KpiMeasurement:
    """4 KPI 산출 — T5.2 scripts/kpi_measure.py에서 호출 + cron에서 호출."""

    # 1) 적재 커버리지 = INDEXED / (INDEXED + DROPPED + ERROR)
    base_q = select(Page).where(Page.status.in_(["INDEXED", "DROPPED", "ERROR"]))
    if domain_id is not None:
        base_q = base_q.where(Page.domain_id == domain_id)

    indexed = await s.scalar(base_q.with_only_columns(func.count()).where(Page.status == "INDEXED"))
    dropped = await s.scalar(base_q.with_only_columns(func.count()).where(Page.status == "DROPPED"))
    errored = await s.scalar(base_q.with_only_columns(func.count()).where(Page.status == "ERROR"))
    total = (indexed or 0) + (dropped or 0) + (errored or 0)
    coverage = Decimal(indexed or 0) / Decimal(total) * 100 if total > 0 else Decimal(0)

    # 2) 적재 신선도 — last_indexed_at - last_fetched_at 평균 (일 단위)
    # posted_at은 본 시스템에서 추출 안 함 — last_fetched_at vs last_indexed_at으로 대체
    freshness_q = select(
        func.avg(extract("epoch", Page.last_indexed_at - Page.last_fetched_at) / 86400.0)
    ).where(Page.status == "INDEXED",
            Page.last_indexed_at.is_not(None),
            Page.last_fetched_at.is_not(None))
    if domain_id is not None:
        freshness_q = freshness_q.where(Page.domain_id == domain_id)
    avg_freshness = await s.scalar(freshness_q)

    # 3) Job 성공률 — 최근 N일
    since = datetime.now(timezone.utc) - timedelta(days=days)
    job_q = select(
        func.count().filter(CrawlJob.status == "SUCCESS"),
        func.count(),
    ).where(CrawlJob.created_at >= since)
    if domain_id is not None:
        job_q = job_q.where(CrawlJob.domain_id == domain_id)
    success, total_jobs = (await s.execute(job_q)).one()
    job_rate = Decimal(success) / Decimal(total_jobs) * 100 if total_jobs > 0 else Decimal(100)

    # 4) 월 LLM 비용 — observability/cost_monitor.py 재사용
    monthly_cost = await get_monthly_cost_krw(s)

    return KpiMeasurement(
        domain_id=domain_id, measured_at=datetime.now(timezone.utc),
        coverage_pct=coverage.quantize(Decimal("0.01")),
        avg_freshness_days=Decimal(avg_freshness or 0).quantize(Decimal("0.01")) if avg_freshness else None,
        job_success_rate_pct=job_rate.quantize(Decimal("0.01")),
        monthly_cost_krw=monthly_cost,
        indexed_count=indexed or 0, dropped_count=dropped or 0, error_count=errored or 0,
    )


async def save_snapshot(s: AsyncSession, m: KpiMeasurement) -> int:
    snap = KpiSnapshot(
        domain_id=m.domain_id, measured_at=m.measured_at,
        coverage_pct=m.coverage_pct, avg_freshness_days=m.avg_freshness_days,
        job_success_rate_pct=m.job_success_rate_pct,
        monthly_cost_krw=m.monthly_cost_krw,
        indexed_count=m.indexed_count, dropped_count=m.dropped_count, error_count=m.error_count,
    )
    s.add(snap); await s.commit(); await s.refresh(snap)
    return snap.id
```

### Daily cron (`tasks/cron_kpi.py`)

```python
async def cron_kpi_daily(ctx) -> None:
    """매일 00:30 KST — 전체 KPI 1행 + 도메인별 KPI N행 snapshot 저장."""
    s = ctx["session_factory"]()

    # 전체
    m = await measure_kpi(s, domain_id=None, days=7)
    await save_snapshot(s, m)
    logger.info("kpi_snapshot_saved", scope="all", coverage=str(m.coverage_pct),
                cost_krw=str(m.monthly_cost_krw))

    # 도메인별
    domains = (await s.execute(select(Domain).where(Domain.is_active == True))).scalars().all()
    for d in domains:
        m = await measure_kpi(s, domain_id=d.id, days=7)
        await save_snapshot(s, m)


# WorkerSettings.cron_jobs에 추가:
cron(cron_kpi_daily, hour={0}, minute={30}, timezone=ZoneInfo("Asia/Seoul"))
```

### KPI API (`api/routers/kpi.py`)

```python
@router.get("/latest", response_model=KpiOut)
async def get_latest_kpi(domain_id: int | None = None, s: AsyncSession = SessionDep, _admin=AdminDep):
    """가장 최근 snapshot 1행. 없으면 측정해서 반환(snapshot 저장 안 함)."""
    snap = await s.scalar(
        select(KpiSnapshot)
        .where(KpiSnapshot.domain_id == domain_id)
        .order_by(KpiSnapshot.measured_at.desc()).limit(1)
    )
    if snap is None:
        m = await measure_kpi(s, domain_id=domain_id)
        return _to_out(m, snapshot=False)
    return _to_out(snap, snapshot=True)


@router.get("/history", response_model=list[KpiOut])
async def get_kpi_history(domain_id: int | None = None, days: int = 30,
                          s: AsyncSession = SessionDep, _admin=AdminDep):
    """최근 N일의 daily snapshot 목록. 대시보드 차트용."""
    since = datetime.now(timezone.utc) - timedelta(days=days)
    rows = (await s.execute(
        select(KpiSnapshot)
        .where(KpiSnapshot.domain_id == domain_id, KpiSnapshot.measured_at >= since)
        .order_by(KpiSnapshot.measured_at)
    )).scalars().all()
    return [_to_out(r, snapshot=True) for r in rows]
```
