### 1. 로그 마스킹 (`observability/masking.py`)

```python
SECRET_KEY_PATTERNS = [
    re.compile(r"(sk-[A-Za-z0-9]{16,})", re.IGNORECASE),                  # OpenAI key
    re.compile(r"(Bearer\s+[A-Za-z0-9._-]{20,})", re.IGNORECASE),         # Bearer token
    re.compile(r"(Basic\s+[A-Za-z0-9+/=]{20,})", re.IGNORECASE),          # Basic auth
    re.compile(r"(postgresql\+asyncpg://[^:]+:)([^@]+)(@)"),               # DSN password
    re.compile(r"(redis://[^:]+:)([^@]+)(@)"),
    re.compile(r"(session=)([A-Za-z0-9._-]{20,})"),                        # session cookie
]


def mask(s: str, *, max_len: int = 8000) -> str:
    """
    1) 시크릿 패턴 → "***" 치환 (DSN/cookie는 password 부분만 마스킹)
    2) 길이 제한 — max_len 초과 시 앞 N자 + "...truncated..."
    """
    if len(s) > max_len:
        s = s[:max_len] + "...truncated..."
    for pat in SECRET_KEY_PATTERNS:
        s = pat.sub(_mask_repl(pat), s)
    return s


def _mask_repl(pat):
    """패턴별 마스킹 — DSN의 password만, cookie의 값만 등."""
```

### 3. 비용 모니터 (`observability/cost_monitor.py`)

```python
async def get_monthly_cost_krw(s: AsyncSession) -> Decimal:
    """이번 달(`Asia/Seoul` 기준) llm_usage 합계 USD × USD_TO_KRW_RATE."""
    today = datetime.now(ZoneInfo("Asia/Seoul"))
    first_of_month = today.replace(day=1, hour=0, minute=0, second=0, microsecond=0)

    cost_usd = await s.scalar(
        select(func.coalesce(func.sum(LlmUsage.cost_usd), 0))
        .where(LlmUsage.created_at >= first_of_month)
    )
    return (Decimal(cost_usd) * Decimal(str(settings.usd_to_krw_rate))).quantize(Decimal("0.01"))


async def check_budget_threshold(s: AsyncSession) -> AlertDecision:
    """80% 임계 도달·해소 판단 + Redis로 1회 발신 보장."""
    cost_krw = await get_monthly_cost_krw(s)
    budget = Decimal(settings.openai_monthly_budget_krw)
    threshold = budget * Decimal("0.8")
    used_pct = float(cost_krw / budget * 100) if budget > 0 else 0.0

    redis = await get_redis()
    month_key = datetime.now(ZoneInfo("Asia/Seoul")).strftime("alert:budget80:%Y%m")
    already_sent = await redis.exists(month_key)

    decision = AlertDecision(
        cost_krw=cost_krw, budget_krw=budget, used_pct=used_pct,
        crossed_threshold=cost_krw >= threshold and not already_sent,
        currently_over=cost_krw >= threshold,
    )

    if decision.crossed_threshold:
        await redis.setex(month_key, 35 * 86400, "1")    # 35일 — 다음 달 reset
    return decision


@dataclass
class AlertDecision:
    cost_krw: Decimal
    budget_krw: Decimal
    used_pct: float
    crossed_threshold: bool        # 알림 발신 필요
    currently_over: bool           # UI 배너 표시 필요
```

### 4. 이메일 발신 (`observability/email.py`)

KSPO 사내 SMTP 가정. 인증 없거나 단순 인증 모두 지원:

```python
async def send_alert_email(*, subject: str, body: str) -> bool:
    """settings.smtp_host/port/from/to_recipients. 실패 시 WARN 로그 + False (알림 발신 자체는 실패해도 cron 실패시키지 않음)."""
    if not settings.smtp_host or not settings.smtp_to_recipients:
        logger.warning("smtp_not_configured — alert skipped")
        return False
    try:
        msg = EmailMessage()
        msg["Subject"] = subject
        msg["From"] = settings.smtp_from
        msg["To"] = ", ".join(settings.smtp_to_recipients.split(","))
        msg.set_content(body)
        async with aiosmtplib.SMTP(hostname=settings.smtp_host, port=settings.smtp_port) as smtp:
            if settings.smtp_user:
                await smtp.login(settings.smtp_user, settings.smtp_password)
            await smtp.send_message(msg)
        logger.info("alert_email_sent", to=settings.smtp_to_recipients, subject=subject)
        return True
    except Exception as e:
        logger.warning("alert_email_failed", error=str(e))
        return False
```

### 5. ARQ cron — 분 단위 임계 검사 (`tasks/cron_alerts.py`)

```python
async def check_budget_alerts(ctx) -> None:
    s = ctx["session_factory"]()
    decision = await check_budget_threshold(s)
    if decision.crossed_threshold:
        await send_alert_email(
            subject=f"[kspo-ingest] OpenAI 월 누계 80% 도달 ({decision.used_pct:.1f}%)",
            body=f"이번 달 OpenAI 누계 비용이 한도의 80%에 도달했습니다.\n"
                 f"누계: {decision.cost_krw}원 / 한도: {decision.budget_krw}원\n"
                 f"런북 B.2.B 참조: docs/runbook/B_openai_failure.md",
        )
        logger.info("budget_alert_sent", cost_krw=str(decision.cost_krw), used_pct=decision.used_pct)
```

### 6. UI 배너 — `GET /api/alerts`

```python
@router.get("/alerts", response_model=AlertsOut)
async def list_active_alerts(s: AsyncSession = SessionDep, _admin=AdminDep):
    """현재 active 알림 — UI 배너 표시용. 30초 폴링."""
    decision = await check_budget_threshold(s)
    alerts = []
    if decision.currently_over:
        alerts.append({
            "kind": "budget_80",
            "severity": "warning",
            "message": f"OpenAI 월 누계 {decision.used_pct:.1f}% — 한도 임박",
            "details": {"cost_krw": str(decision.cost_krw), "budget_krw": str(decision.budget_krw)},
        })
    # 향후 KB 차원 mismatch, OpenAI 인증 실패도 여기서 조회
    return {"alerts": alerts}
```

### 테스트

`test_logging_masking.py` (≥8):
- OpenAI 키 → `***`
- Bearer 토큰 → 마스킹
- Basic auth → 마스킹
- DSN password → 마스킹 (host:port는 보존)
- Redis URL password → 마스킹
- session 쿠키 값 → 마스킹
- 길이 8000자 초과 → truncated
- 정상 메시지 → 변경 없음

`test_cost_monitor.py` (≥6):
- 이번 달 llm_usage 0행 → 0원
- 이번 달 5행 합계 (Decimal 정확) → 정확한 KRW
- 80% 미만 → crossed_threshold=False
- 80% 도달 첫 호출 → crossed_threshold=True, Redis 키 set
- 80% 도달 후 두 번째 호출 → crossed_threshold=False (이미 발신)
- 다른 달의 llm_usage는 미포함

`test_alert_email.py` (≥4):
- SMTP 정상 → True + log
- SMTP 미설정 → False + WARN 로그
- SMTP 호출 실패 → False (예외 raise 안 함)
- 수신자 다수 (콤마) 정확 파싱

`test_alerts_api.py` (≥4):
- 누계 0원 → alerts=[]
- 80% 초과 → alerts=[{kind:'budget_80', ...}]
- 인증 미충족 → 401
- 30초 폴링 시뮬레이션 (응답 시간 < 100ms)
